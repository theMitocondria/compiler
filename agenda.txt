	first_container = cppc.front  //isk mtlb ye last time sabse late execute hua hoga

	resp = first_container(code , input)

		mutex.lovk()

compile_code(code){
	while(in_execution == 5) ;
	mutex.lock();
	in_execution += 1;
	mutex.unlock()
	response ;
	switch (lang):

		case java :
			response = java(body)
			break;

			response = py(body)
			break;

}
